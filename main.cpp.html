#include <iostream>
using namespace std;
class bankAccount {
private:
    int acc_num;
    string acc_name;
    float balance;
public:
    bankAccount(int num, string name, float  intalize_balance= 0.0);
    int get_acc_num();
    string get_acc_name();
    float get_balance();

    void deposit_money(float amount);
    void withdraw_money(float amount);

};


bankAccount::bankAccount(int num, string name, float intalize_balance )
{
    acc_num = num;
    acc_name = name;
    balance = intalize_balance;
}

int bankAccount::get_acc_num()
{
    return acc_num;
}
string bankAccount::get_acc_name()
{
    return acc_name;
}
float bankAccount::get_balance()
{
    return balance;
}
void bankAccount::deposit_money(float amount) {
    if (amount > 0) {
        balance += amount;
        cout << "Succesfully depost process" << endl;
    }
    else {
        cout << "Cannot deposit negative amount " << endl;
    }
}
void bankAccount::withdraw_money(float amount)
{
    if (amount <= 0) {
        cout << "The amount of money must be greater tan zero" << endl;
    }
    else if (amount > balance) {
        cout << "Insufficent money in the account" << endl;
    }
    else {
        balance -= amount;
        cout << "Succesed withdraw process!" << endl;
    }
}


int main() {
    bankAccount client1( 221 , "ahmed omar" , 6000);
    bankAccount client2(222, "adel yasser", 4000);

    client1.deposit_money(400); // add money
    client2.deposit_money(-20.5); // error ----> will give hime sign "cannot deposit negative amount "
    client1.withdraw_money(300);// withdraw money
    client2.withdraw_money(7000);// error ----> will give hime sign " insufficent money in the account"

    cout << "----------------INFORMATIONS------------"<<endl;
    cout << "Client 1 Name: " <<client1.get_acc_name()<<endl;
    cout <<"Client 1 Acc Number: " << client1.get_acc_num()<<endl;
    cout << "Client 1 balance : " <<client1.get_balance()<<endl;
    cout << "Client 2 Name: " << client2.get_acc_name()<<endl;
    cout<< "Client 2 Acc Number: " << client2.get_acc_num()<<endl;
    cout << "Client 2 Balance: " << client2.get_balance()<<endl;
}
